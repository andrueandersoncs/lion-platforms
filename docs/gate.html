// 1. Runtime check before any tool executes (7 ms P50)
const decision = await fetch("https://api.liongate.dev/v1/check", {
  method: "POST",
  headers: { Authorization: `Bearer ${process.env.LION_GATE_KEY}` },
  body: JSON.stringify({ policy: "refunds@v4", action: toolCall, facts: liveAccount })
}).then((r) => r.json());

if (!decision.allowed) return routeToApproval(decision.deniedReasons);
